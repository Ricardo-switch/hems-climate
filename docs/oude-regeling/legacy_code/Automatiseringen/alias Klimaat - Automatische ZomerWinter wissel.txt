alias: Klimaat - Automatische Zomer/Winter wissel
description: >-
  Schakelt de zomer/winter layout en zet de strategie veilig op PV koelen of
  Uit.
triggers:
  - id: zomer_modus
    trigger: numeric_state
    entity_id: sensor.temperatuur
    above: 20
    for:
      minutes: 10
  - id: zomer_modus_zon_door_raam
    trigger: numeric_state
    entity_id: sensor.beneden_temperatuur
    above: 22.5
    for:
      minutes: 30
  - id: winter_modus
    trigger: numeric_state
    entity_id: sensor.beneden_temperatuur
    below: 21
    for:
      minutes: 30
actions:
  - choose:
      - conditions:
          - condition: trigger
            id:
              - zomer_modus
              - zomer_modus_zon_door_raam
        sequence:
          - action: input_select.select_option
            target:
              entity_id: input_select.koel_strategie
            data:
              option: Pv koelen
          - action: input_boolean.turn_on
            target:
              entity_id: input_boolean.zomer_modus_actief
          - action: input_boolean.turn_off
            target:
              entity_id: input_boolean.pv_verwarmen_actief
      - conditions:
          - condition: trigger
            id: winter_modus
          - condition: template
            value_template: '{{ states(''input_select.koel_strategie'') != ''Handmatig koelen'' }}'
        sequence:
          - action: input_select.select_option
            target:
              entity_id: input_select.koel_strategie
            data:
              option: Uit
          - action: input_boolean.turn_off
            target:
              entity_id: input_boolean.zomer_modus_actief
          - action: input_boolean.turn_on
            target:
              entity_id: input_boolean.pv_verwarmen_actief
mode: single
