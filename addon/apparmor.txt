#include <tunables/global>

# Het AppArmor-profiel van SunArc (besluit 236).
#
# Een eigen profiel geeft de add-on het beveiligingscijfer 8, het hoogste. De
# Supervisor laadt het bij het installeren en zet de naam hieronder om naar de
# volledige naam van de add-on.
#
# Twee lagen. Het opstartscript (run.sh, met bashio) mag ruim, net als in het
# voorbeeld van Home Assistant: het leest de opties en kopieert het wekwoord.
# De dienst zelf, node, draait in een eigen, smal profiel: lezen wat in het
# image staat, schrijven alleen in de mappen die config.yaml aankoppelt, en het
# netwerk voor Home Assistant en het weerbericht.
#
# Nu in de leerstand (`complain`): niets wordt tegengehouden, elke stap buiten
# het profiel komt in het logboek van de host. Lees dat na een dag met
#   journalctl _TRANSPORT=audit -g 'apparmor="ALLOWED"'
# of `dmesg | grep apparmor`, vul aan wat ontbreekt, en haal dan `complain`
# weg (de tweede stap van besluit 236).

profile hems_climate flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>

  # Het opstartscript
  file,
  network,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay en bashio uit de basis van Home Assistant
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /run/{,**} rwk,
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /dev/tty rw,

  # De dienst in zijn eigen profiel
  /usr/bin/node cx -> node,

  profile node flags=(attach_disconnected,mediate_deleted,complain) {
    #include <abstractions/base>

    # Stoppen en opnieuw starten via run.sh
    signal (receive) peer=*_hems_climate,
    signal (receive) peer=hems_climate,

    # Home Assistant, de Supervisor, het weerbericht, de stem (Wyoming)
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,

    # Node zelf en de code in het image
    /usr/bin/node mrix,
    /usr/lib/** mr,
    /lib/** mr,
    /app/** r,

    # Wat node leest om te draaien: certificaten, namen, tijdzones, het systeem
    /etc/** r,
    /usr/share/zoneinfo/** r,
    /usr/share/ca-certificates/** r,
    /proc/** r,
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    /dev/urandom r,
    /dev/null rw,
    /tmp/** rwk,

    # De mappen uit config.yaml: instellingen, het logboek, de opties, en de
    # integratie in de configuratie van Home Assistant
    /config/{,**} rwk,
    /share/{,**} rwk,
    /data/{,**} rwk,
    /homeassistant/{,**} rwk,
  }
}
