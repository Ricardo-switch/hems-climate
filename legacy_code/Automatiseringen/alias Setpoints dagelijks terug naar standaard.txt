alias Setpoints dagelijks terug naar standaard
description -
  Reset setpoints naar standaard en herstelt de koel-strategie (HandmatigAuto)
  netjes naar de basis.
triggers
  - at '030000'
    trigger time

  - if
      - condition template
        value_template -
          {{ states('input_select.koel_strategie') in ['Handmatig koelen',
          'Automatisch koelen'] }}
    then

          - condition state
            entity_id input_boolean.zomer_modus_actief
            state 'on'
        then

            target
              entity_id input_select.koel_strategie
            data
              option Pv koelen

            target
              entity_id input_select.koel_strategie
            data

      - conditions

            value_template -
              {{ states('input_number.temp_airco_woonkamer')float(0)
